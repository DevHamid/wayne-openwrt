# Appended to vendor/wayne-oss-perf_defconfig at build time.

# --- devtmpfs auto-mount: OpenWrt's procd expects /dev populated by kernel ---
# NOTE (verified in drivers/base/Kconfig): DEVTMPFS_MOUNT does NOT apply during
# initramfs boot -- the help text says "does not affect initramfs based booting,
# here the devtmpfs filesystem always needs to be mounted manually". That is why
# BOTH our /init scripts run `mount -t devtmpfs ...` explicitly. We set it
# anyway so a later boot-from-disk rootfs also self-populates /dev.
CONFIG_DEVTMPFS=y
CONFIG_DEVTMPFS_MOUNT=y

# tmpfs: used by OpenWrt for /tmp and by the initramfs.

CONFIG_TMPFS_POSIX_ACL=y

# NOTE: CONFIG_NETFILTER_XT_TARGET_LOG is intentionally NOT set. This LKGeek tree
# only builds the nf_log core (nf_log.o) when DEBUG_KERNEL=y (net/netfilter/Makefile
# line 4), and the vendor defconfig leaves DEBUG_KERNEL off. Enabling XT_TARGET_LOG
# compiles nf_log_ipv4/ipv6/xt_LOG which reference missing nf_log_* symbols and
# fails the final vmlinux link. Add both (LOG + DEBUG_KERNEL) only if firewall
# LOG rules are ever needed.
CONFIG_NETFILTER_XT_TARGET_MASQUERADE=y

# IPv6: OpenWrt defaults to dual-stack.
CONFIG_IPV6=y

# Bridge & VLAN: router plumbing.
CONFIG_BRIDGE=y
CONFIG_VLAN_8021Q=y

# USB gadget ethernet: ECM (CDC Ethernet) is the PRIMARY transport.
# SWITCHED FROM RNDIS (2026-11-07): RNDIS stalled hard on a Linux host --
# rndis_host logged "NETDEV WATCHDOG: transmit queue 0 timed out", ARP never
# resolved, 100% packet loss. RNDIS is a Microsoft protocol with known
# instability on non-Windows hosts; ECM is the native Linux CDC path
# (cdc_ether driver, no rndis_host quirks). The PC needs no extra driver.
CONFIG_USB_CONFIGFS=y
CONFIG_USB_CONFIGFS_ECM=y
# NCM kept as a fallback (also composable, higher throughput).
CONFIG_USB_CONFIGFS_NCM=y
# RNDIS kept enabled in the kernel but NOT used by the gadget -- Windows
# hosts that fail ECM can still be served by switching the function back.
CONFIG_USB_CONFIGFS_RNDIS=y
CONFIG_USB_GADGET=y

# USB serial console (CDC ACM) -- the debug console we need.
# USB_G_SERIAL removed (M5 finding): single UDC cannot hold both the legacy
# g_serial and a configfs composite gadget -- g_serial binds at boot and the
# configfs gadget then gets 'Resource busy'. Console + ethernet come from ONE
# configfs composite gadget (acm.usb0 + rndis.usb0) built by the /init scripts.
# CONFIG_USB_G_SERIAL is not set
CONFIG_USB_CONFIGFS_ACM=y
CONFIG_USB_LIBCOMPOSITE=y

# eMMC/block (already vendor-enabled, re-affirmed).
CONFIG_MMC=y
CONFIG_MMC_BLOCK=y

# Filesystems OpenWrt's flash tools and containers may want.
CONFIG_OVERLAY_FS=y
CONFIG_SQUASHFS=y
CONFIG_EXT4_FS=y

# Compression for initramfs (gzip is the standard).
CONFIG_RD_GZIP=y

# Kernel config for the debug shell (minimal).
# These are already in the vendor defconfig but re-affirmed for clarity.
CONFIG_BLK_DEV_INITRD=y

# ===========================================================================
# v0.8 -- USB OTG HOST mode + permanent rootfs enablers (Course A)
# ===========================================================================
# GOAL: let the phone act as a USB *host* so a USB-OTG stick can hold the
# persistent rootfs (root=/dev/sda1), and so USB ethernet works later (M5).
#
# WHY THESE ARE NEEDED (verified against THIS tree, not from memory):
#   drivers/usb/dwc3/Kconfig:  USB_DWC3_DUAL_ROLE depends on EXTCON=y
#     => EXTCON is MANDATORY, not optional.
#   Vendor defconfig sets NO dwc3 mode, so the `choice` falls back to its
#   default. We now state DUAL_ROLE explicitly, so host is available while the
#   g_serial console (device role) keeps working.

# extcon core -- the external-connector class DWC3 dual-role depends on.
CONFIG_EXTCON=y

# PMIC cable/role detection for Qualcomm PMICs (wayne = PM660/PM660L).
# "Qualcomm USB extcon support ... SPMI PMIC based USB cable detection".
CONFIG_EXTCON_QCOM_SPMI_MISC=y

# DWC3 dual role: BOTH host and gadget from the same controller.
# This is what makes /dev/sda (a USB stick) appear while ttyGS0 stays alive.
CONFIG_USB_DWC3_DUAL_ROLE=y

# USB core OTG state machine (USB_OTG lives in drivers/usb/core/Kconfig).
CONFIG_USB_OTG=y

# Type-C port-controller manager; selects USB_ROLE_SWITCH.
CONFIG_TYPEC_TCPM=y

# Host-mode plumbing. USB_XHCI_HCD + USB_STORAGE + SCSI already =y in the
# vendor defconfig (lines 607/608/375) -- re-affirmed so a defconfig change
# can never silently drop our ability to read a USB disk.
CONFIG_USB_XHCI_HCD=y
CONFIG_USB_STORAGE=y
CONFIG_SCSI=y
CONFIG_BLK_DEV_SD=y

# USB Attached SCSI (faster bulk storage than the old usb-storage path).
CONFIG_USB_UAS=y

# CDC-ACM as a HOST driver is part of USB_ACM (shared with the gadget side).
CONFIG_USB_ACM=y

# NOTE on the gadget side: USB_CONFIGFS_MASS_STORAGE (already =y in the vendor
# defconfig) would let the phone EXPOSE a disk to a PC. That is a different
# feature from reading a USB stick and is left as-is.
# The legacy USB_MASS_STORAGE (gadget) is deliberately NOT set: it shares the
# "precomposed gadgets" single-select choice with USB_G_SERIAL, and setting
# both makes kconfig silently drop one -- the same trap documented above.

# ===========================================================================
# v0.9.10 -- M7 Docker prerequisites (open probe)
# ===========================================================================
# AUDIT (2026-10-17) against the vendor defconfig + this fragment:
#   present : CGROUPS, CGROUP_CPUACCT, CGROUP_SCHED, NAMESPACES, NET_NS,
#             OVERLAY_FS, BRIDGE, NF_NAT, NETFILTER_XT_TARGET_MASQUERADE
#   MISSING : MEMCG, CGROUP_DEVICE, CGROUP_FREEZER, CGROUP_PIDS, VETH
# dockerd/runc refuse to start without a memory cgroup controller
# ("Your kernel does not support memory limit capabilities"), and containers
# get no network without veth pairs. These are pure enablers: nothing above
# changes existing boot behaviour.
# MEMCG was withheld as a bisect probe on the 12b01ae978 pin (run f6613d1
# failed the build). Resolved: LKGeek_sdm660 commit 1c46e3f2 on
# fix-ksu-sugrant landed the working container set and is only on that branch,
# so wireguard-linux does not carry it. The failure was never the vendor mm/
# tree failing to compile with MEMCG -- it was MEMCG_KMEM without the MGLRU
# shrinker_rwsem fix. Docker's memory limits and OOM behaviour need MEMCG, and
# MGLRU's per-memcg reclaim paths (get_lruvec, get_mm_list,
# mem_cgroup_page_lruvec) are all #ifdef CONFIG_MEMCG guarded. MEMCG_KMEM is a
# hidden bool with 'default y' under MEMCG, pulled in automatically.
CONFIG_MEMCG=y
CONFIG_MEMCG_SWAP=y
CONFIG_MEMCG_SWAP_ENABLED=y
CONFIG_CGROUP_DEVICE=y
CONFIG_CGROUP_FREEZER=y
CONFIG_CGROUP_PIDS=y
CONFIG_CGROUP_SCHED=y
CONFIG_VETH=y
# Network namespaces are already on; the veth peer needs the bridge module
# only when docker0 bridging is used (it is the default driver).
CONFIG_BRIDGE=y
# CONFIG_BRIDGE_NETFILTER is deliberately NOT set. Proven 2026-10-17: this
# vendor tree's net/bridge/br_private.h has no nf_call_iptables / _ip6tables /
# _arptables members, so br_netlink.c fails to compile:
#   error: no member named 'nf_call_iptables' in 'struct net_bridge'
# LKGeek dropped br_netfilter. Consequence for M7: docker0 bridging still works,
# but iptables never sees bridged frames, so net.bridge.bridge-nf-call-iptables
# is a no-op and published-port DNAT across the bridge needs the userland proxy
# (docker --userland-proxy=true, its default) to hold. Record it as a probe
# limitation, do not re-add this symbol without porting br_netfilter.

# ---------------------------------------------------------------------------
# Loadable module support. REQUIRED for modem_boot.ko / rmtfs.ko: this tree's
# vendor defconfig carries no CONFIG_MODULES line, so kconfig defaults it to
# n and 'make modules' / 'make modules_install' abort with:
#   "The present kernel configuration has modules disabled."
# =y is the only safe value (MODULES=y alone leaves MODVERSIONS etc. off; the
# Android-style '=m' would turn the whole vendor tree into a module tree).
CONFIG_MODULES=y
