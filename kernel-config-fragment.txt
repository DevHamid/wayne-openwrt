# Kernel config additions for OpenWrt userspace.
# Appended to vendor/wayne-oss-perf_defconfig at build time.
# Each option exists for a specific OpenWrt need; do not remove without checking.

# --- devtmpfs auto-mount: OpenWrt's procd expects /dev populated by kernel ---
# NOTE (verified in drivers/base/Kconfig): DEVTMPFS_MOUNT does NOT apply during
# initramfs boot -- the help text says "does not affect initramfs based booting,
# here the devtmpfs filesystem always needs to be mounted manually". That is why
# BOTH our /init scripts run `mount -t devtmpfs ...` explicitly. We set it
# anyway so a later boot-from-disk rootfs also self-populates /dev.
CONFIG_DEVTMPFS=y
CONFIG_DEVTMPFS_MOUNT=y

# tmpfs: used by OpenWrt for /tmp and by the initramfs.

CONFIG_TMPFS_POSIX_ACL=y

# NOTE: CONFIG_NETFILTER_XT_TARGET_LOG is intentionally NOT set. This LKGeek tree
# only builds the nf_log core (nf_log.o) when DEBUG_KERNEL=y (net/netfilter/Makefile
# line 4), and the vendor defconfig leaves DEBUG_KERNEL off. Enabling XT_TARGET_LOG
# compiles nf_log_ipv4/ipv6/xt_LOG which reference missing nf_log_* symbols and
# fails the final vmlinux link. Add both (LOG + DEBUG_KERNEL) only if firewall
# LOG rules are ever needed.
CONFIG_NETFILTER_XT_TARGET_MASQUERADE=y

# Bridge & VLAN: router plumbing.

# USB gadget ethernet (NCM/RNDIS via configfs): already in the vendor defconfig.
# NOTE: we deliberately do NOT set CONFIG_USB_ETH here. The legacy "precomposed"
# gadgets (g_ether/g_serial/...) form a single-select Kconfig choice -- setting
# both USB_ETH and USB_G_SERIAL makes kconfig keep one and silently drop the
# other with an "override:" warning. Future USB-ethernet work should use the
# configfs functions below (NCM/RNDIS), which are composable at runtime.
CONFIG_USB_CONFIGFS=y
CONFIG_USB_CONFIGFS_NCM=y
CONFIG_USB_CONFIGFS_RNDIS=y
CONFIG_USB_GADGET=y

# USB serial console (CDC ACM) -- the debug console we need.
# USB_G_SERIAL removed (M5 finding): single UDC cannot hold both the legacy
# g_serial and a configfs composite gadget -- g_serial binds at boot and the
# configfs gadget then gets 'Resource busy'. Console + ethernet come from ONE
# configfs composite gadget (acm.usb0 + rndis.usb0) built by the /init scripts.
# CONFIG_USB_G_SERIAL is not set
CONFIG_USB_CONFIGFS_ACM=y
CONFIG_USB_LIBCOMPOSITE=y

# eMMC/block (already vendor-enabled, re-affirmed).
CONFIG_MMC=y
CONFIG_MMC_BLOCK=y

# Filesystems OpenWrt's flash tools and containers may want.

CONFIG_SQUASHFS=y
CONFIG_EXT4_FS=y

# Kernel config for the debug shell (minimal).

CONFIG_BLK_DEV_INITRD=y

# ===========================================================================
# v0.8 -- USB OTG HOST mode + permanent rootfs enablers (Course A)
# ===========================================================================
# GOAL: let the phone act as a USB *host* so a USB-OTG stick can hold the
# persistent rootfs (root=/dev/sda1), and so USB ethernet works later (M5).
#
# WHY THESE ARE NEEDED (verified against THIS tree, not from memory):
#   drivers/usb/dwc3/Kconfig:  USB_DWC3_DUAL_ROLE depends on EXTCON=y
#     => EXTCON is MANDATORY, not optional.
#   Vendor defconfig sets NO dwc3 mode, so the `choice` falls back to its
#   default. We now state DUAL_ROLE explicitly, so host is available while the
#   g_serial console (device role) keeps working.

# extcon core -- the external-connector class DWC3 dual-role depends on.
CONFIG_EXTCON=y

# PMIC cable/role detection for Qualcomm PMICs (wayne = PM660/PM660L).
# "Qualcomm USB extcon support ... SPMI PMIC based USB cable detection".
CONFIG_EXTCON_QCOM_SPMI_MISC=y

# DWC3 dual role: BOTH host and gadget from the same controller.
# This is what makes /dev/sda (a USB stick) appear while ttyGS0 stays alive.
CONFIG_USB_DWC3_DUAL_ROLE=y

# USB core OTG state machine (USB_OTG lives in drivers/usb/core/Kconfig).
CONFIG_USB_OTG=y

# Type-C port-controller manager; selects USB_ROLE_SWITCH.
CONFIG_TYPEC_TCPM=y

# Host-mode plumbing. USB_XHCI_HCD + USB_STORAGE + SCSI already =y in the
# vendor defconfig (lines 607/608/375) -- re-affirmed so a defconfig change
# can never silently drop our ability to read a USB disk.
CONFIG_USB_XHCI_HCD=y
CONFIG_USB_STORAGE=y
CONFIG_SCSI=y
CONFIG_BLK_DEV_SD=y

# USB Attached SCSI (faster bulk storage than the old usb-storage path).
CONFIG_USB_UAS=y

# CDC-ACM as a HOST driver is part of USB_ACM (shared with the gadget side).
CONFIG_USB_ACM=y

# NOTE on the gadget side: USB_CONFIGFS_MASS_STORAGE (already =y in the vendor
# defconfig) would let the phone EXPOSE a disk to a PC. That is a different
# feature from reading a USB stick and is left as-is.
# The legacy USB_MASS_STORAGE (gadget) is deliberately NOT set: it shares the
# "precomposed gadgets" single-select choice with USB_G_SERIAL, and setting
# both makes kconfig silently drop one -- the same trap documented above.

# AUDIT (2026-10-17) against the vendor defconfig + this fragment:
#   present : CGROUPS, CGROUP_CPUACCT, CGROUP_SCHED, NAMESPACES, NET_NS,
#             OVERLAY_FS, BRIDGE, NF_NAT, NETFILTER_XT_TARGET_MASQUERADE
#   MISSING : MEMCG, CGROUP_DEVICE, CGROUP_FREEZER, CGROUP_PIDS, VETH
# dockerd/runc refuse to start without a memory cgroup controller
# ("Your kernel does not support memory limit capabilities"), and containers
# get no network without veth pairs. These are pure enablers: nothing above
# changes existing boot behaviour.
# MEMCG deliberately withheld in this commit: run f6613d1 (which failed the
# kernel build) without it to bisect. If this build is green, MEMCG is the
# breaker and the vendor mm/ tree does not compile with it.
# CONFIG_MEMCG is not set
CONFIG_CGROUP_DEVICE=y
CONFIG_CGROUP_FREEZER=y
CONFIG_CGROUP_PIDS=y
CONFIG_CGROUP_SCHED=y
CONFIG_VETH=y
# Network namespaces are already on; the veth peer needs the bridge module
# only when docker0 bridging is used (it is the default driver).
CONFIG_BRIDGE=y
# CONFIG_BRIDGE_NETFILTER is deliberately NOT set. Proven 2026-10-17: this
# vendor tree's net/bridge/br_private.h has no nf_call_iptables / _ip6tables /
# _arptables members, so br_netlink.c fails to compile:
#   error: no member named 'nf_call_iptables' in 'struct net_bridge'
# LKGeek dropped br_netfilter. Consequence for M7: docker0 bridging still works,
# but iptables never sees bridged frames, so net.bridge.bridge-nf-call-iptables
# is a no-op and published-port DNAT across the bridge needs the userland proxy
# (docker --userland-proxy=true, its default) to hold. Record it as a probe
# limitation, do not re-add this symbol without porting br_netfilter.

# --- WLAN (wlan0 never appears): make the icnss/QMI chain observable ---------
# Verified on-device 2026-11-01 by scripts/wayne-diag.sh (246-line report):
#   icnss: Platform driver probed successfully
#   wlan: Loading driver v5.2.05.23C
#   wlan_hdd_state wlan major(502) initialized
#   ...then NOTHING. No "WLAN FW is ready", no icnss_qmi output at all.
# That silence is not proof of health: DYNAMIC_DEBUG is off (default n,
# lib/Kconfig.debug:81), so every icnss_pr_dbg()/icnss_qmi_pr_dbg() is
# compiled out. The suppressed messages are exactly the ones that mark where
# the chain stops.
#
# RETRACTED (was wrong, caught by that same report): "DEBUG_FS is off, we are
# debugging blind" and the CONFIG_DEBUG_FS=y that followed from it. The running
# kernel ALREADY has CONFIG_DEBUG_FS=y -- it is pulled in transitively by a
# dependency (select DEBUG_FS in mm/Kconfig:692, kernel/trace/Kconfig:471,
# net/sunrpc/Kconfig:40, drivers/md/bcache/Kconfig:23, drivers/edac/Kconfig:36),
# and it is in NEITHER wayne defconfig. The earlier
# "nonexistent directory" failure was an older v0.9.x build, not this one.
# Do NOT add CONFIG_DEBUG_FS -- it is already on.
CONFIG_DYNAMIC_DEBUG=y

# CONFIG_ICNSS_DEBUG is deliberately NOT set. drivers/soc/qcom/Kconfig:965
# depends on DEBUG_FS and its help text says it enables BUG_ON on hardware
# register access -- a hang risk on a device whose WLAN registers may already
# be wedged. DYNAMIC_DEBUG gives us the prints without the BUG_ON.

# --- WLAN: rmtfs_mem driver (candidate, NOT confirmed) -----------------------
# The real failure is the QMI service locator timing out:
#   servloc: init_service_locator: wait for locator service timed out
#   servloc: pd_locator_work: Unable to connect to service locator!, rc = -62
# rc=-62 is -ETIME from service-locator.c:283 (LOCATOR_SERVICE_TIMEOUT=300000ms
# = 300s), NOT ETIMEDOUT(110). The timeout printed at t=313.65s = 300s plus
# ~13s of boot before the workqueue was scheduled, so it ran the full course
# and the remote never answered once.
# With the locator dead: icnss_pd_restart_enable() ->
# get_service_location("wlan","wlan/fw",...) fails (icnss.c:2012) -> no PDR
# notifier -> WLFW_SERVICE_ID_V01 never seen -> wlfw_new_server() never fires
# -> ICNSS_FW_READY never set -> icnss_driver_event_register_driver() bails
# (icnss.c:1339, "FW is not ready yet") -> pld_snoc_probe never runs ->
# no wiphy -> no wlan0.
# adsprpc's "PDR notifier locator for adsp is down" is the SAME locator failing
# for a different consumer, so this is transport-wide, not a WLAN bug.
#
# RETRACTED: "the icnss2 driver can fix this via SKIP_QMI." The wayne-diag
# report shows icnss2 is ABSENT from the running kernel, and
# drivers/soc/qcom/Makefile:106 gates it on CONFIG_ICNSS2, which is not set in
# either defconfig. Meanwhile icnss2/main.c:3533 also matches "qcom,icnss",
# the same compatible as the bound icnss node -- so enabling ICNSS2 would race
# the currently-probing ICNSS for the same DT node. Not a safe swap on
# speculation; do not propose it again without a tested reference.
#
# RETRACTED: "no subsys-pil-tz line for modem is an asymmetry worth chasing."
# subsys-pil-tz.c:1090 takes its subsys name from qcom,firmware-name, and only
# the TZ-based nodes (venus/adsp/cdsp/kgsl) use that driver at all. The modem
# uses a DIFFERENT driver, pil-q6v5-mss (compatible "qcom,pil-q6v55-mss"), so
# its absence from that list is expected and means nothing.
#
# REMAINING OPEN (kernel side is exonerated): nothing in this tree ever calls
# subsys_boot(); on Android userspace triggers the Q6 bring-up. The kernel is
# doing its job. Also note /lib/firmware/ was EMPTY on-device -- the M8
# firmware staging did not persist, a second independent problem.
#
# EXPECT THIS TO BE INSUFFICIENT ON ITS OWN. sdm660.dtsi:1579 already has a
# rmtfs node, but it is wrong for this driver:
#   qcom,rmtfs_sharedmem@0 {
#           compatible = "qcom,sharedmem-uio";   <- binds msm_sharedmem.c:221
#           reg = <0x0 0x200000>;                <- no base address
#           qcom,client-id = <0x00000001>;       <- 1, correct for rmtfs
#   }
# It also sits in the soc block, not reserved-memory. So enabling
# CONFIG_QCOM_RMTFS_MEM compiles a driver that has nothing to bind to.
# A correct node needs compatible "qcom,rmtfs-mem", a real base+size reg, and
# no-map, per arch/arm64/boot/dts/vendor/bindings/reserved-memory/
# qcom,rmtfs-mem.txt. No SDM660 reference for that node exists in any local
# tree (kc419 / wayne-v5 only carry msm8916 and msm8996 examples), so do NOT
# invent a region blind -- find a genuinely free one first. Reserved memory
# map for reference: wlan_msa 0x85700000+0x100000, smem 0x86000000+0x200000,
# removed_regions 0x85800000+0x800000, removed_regions1 0x86200000+0x2d00000.
CONFIG_QCOM_RMTFS_MEM=y
