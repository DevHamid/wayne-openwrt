# Kernel config additions for OpenWrt userspace.
# Appended to vendor/wayne-oss-perf_defconfig at build time.
# Each option exists for a specific OpenWrt need; do not remove without checking.

# --- devtmpfs auto-mount: OpenWrt's procd expects /dev populated by kernel ---
# NOTE (verified in drivers/base/Kconfig): DEVTMPFS_MOUNT does NOT apply during
# initramfs boot -- the help text says "does not affect initramfs based booting,
# here the devtmpfs filesystem always needs to be mounted manually". That is why
# BOTH our /init scripts run `mount -t devtmpfs ...` explicitly. We set it
# anyway so a later boot-from-disk rootfs also self-populates /dev.
CONFIG_DEVTMPFS=y
CONFIG_DEVTMPFS_MOUNT=y

# tmpfs: used by OpenWrt for /tmp and by the initramfs.
CONFIG_TMPFS=y
CONFIG_TMPFS_POSIX_ACL=y

# POSIX message queues: OpenWrt's ubus/procd inter-process messaging needs this.
CONFIG_POSIX_MQUEUE=y

# TUN/TAP: every VPN package in OpenWrt needs this.
CONFIG_TUN=y

# Namespaces & cgroups: procd's service jails and container tools.
CONFIG_NAMESPACES=y
CONFIG_UTS_NS=y
CONFIG_IPC_NS=y
CONFIG_USER_NS=y
CONFIG_PID_NS=y
CONFIG_NET_NS=y
CONFIG_CGROUPS=y
CONFIG_CGROUP_CPUACCT=y
CONFIG_CGROUP_DEVICE=y
CONFIG_CGROUP_FREEZER=y
CONFIG_CGROUP_SCHED=y
CONFIG_CGROUP_PIDS=y

# Netfilter: the iptables-based firewall (OpenWrt 21.02 uses firewall3+iptables).
CONFIG_NETFILTER=y
CONFIG_NF_CONNTRACK=y
CONFIG_NF_TABLES=y
CONFIG_NF_NAT=y
CONFIG_NETFILTER_XTABLES=y
CONFIG_NETFILTER_XT_MATCH_CONNTRACK=y
# NOTE: CONFIG_NETFILTER_XT_TARGET_LOG is intentionally NOT set. This LKGeek tree
# only builds the nf_log core (nf_log.o) when DEBUG_KERNEL=y (net/netfilter/Makefile
# line 4), and the vendor defconfig leaves DEBUG_KERNEL off. Enabling XT_TARGET_LOG
# compiles nf_log_ipv4/ipv6/xt_LOG which reference missing nf_log_* symbols and
# fails the final vmlinux link. Add both (LOG + DEBUG_KERNEL) only if firewall
# LOG rules are ever needed.
CONFIG_NETFILTER_XT_TARGET_MASQUERADE=y

# IPv6: OpenWrt defaults to dual-stack.
CONFIG_IPV6=y

# Bridge & VLAN: router plumbing.
CONFIG_BRIDGE=y
CONFIG_VLAN_8021Q=y

# USB gadget ethernet (NCM/RNDIS via configfs): already in the vendor defconfig.
# NOTE: we deliberately do NOT set CONFIG_USB_ETH here. The legacy "precomposed"
# gadgets (g_ether/g_serial/...) form a single-select Kconfig choice -- setting
# both USB_ETH and USB_G_SERIAL makes kconfig keep one and silently drop the
# other with an "override:" warning. Future USB-ethernet work should use the
# configfs functions below (NCM/RNDIS), which are composable at runtime.
CONFIG_USB_CONFIGFS=y
CONFIG_USB_CONFIGFS_NCM=y
CONFIG_USB_CONFIGFS_RNDIS=y
CONFIG_USB_GADGET=y

# USB serial console (CDC ACM) -- the debug console we need.
# Neither option is in the vendor defconfig; verified against gadget Kconfig:
#  - USB_G_SERIAL selects USB_U_SERIAL + USB_F_ACM + USB_F_SERIAL + USB_LIBCOMPOSITE
#    and produces /dev/ttyGS0 automatically (legacy gadget).
#  - USB_CONFIGFS_ACM also selects USB_U_SERIAL + USB_F_ACM, so the SAME
#    /dev/ttyGS0 appears when bound through configfs (the /init scripts bind it).
# We enable both: legacy g_serial is the zero-config fallback, configfs-ACM is
# the clean path and matches the configfs-NCM/RNDIS functions already enabled.
CONFIG_USB_G_SERIAL=y
CONFIG_USB_CONFIGFS_ACM=y
CONFIG_USB_LIBCOMPOSITE=y

# eMMC/block (already vendor-enabled, re-affirmed).
CONFIG_MMC=y
CONFIG_MMC_BLOCK=y

# Filesystems OpenWrt's flash tools and containers may want.
CONFIG_OVERLAY_FS=y
CONFIG_SQUASHFS=y
CONFIG_EXT4_FS=y

# Compression for initramfs (gzip is the standard).
CONFIG_RD_GZIP=y

# Kernel config for the debug shell (minimal).
# These are already in the vendor defconfig but re-affirmed for clarity.
CONFIG_BLK_DEV_INITRD=y
